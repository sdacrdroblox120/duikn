-- ===== 通用组件工具 =====

-- 圆角
local function corner(parent, r)
    local o = Instance.new("UICorner")
    o.CornerRadius = UDim.new(0, r or 8)
    o.Parent = parent
    return o
end

-- 描边
local function stroke(parent, color, thick, trans)
    local o = Instance.new("UIStroke")
    o.Color = color or THEME.line
    o.Thickness = thick or 1
    o.Transparency = trans or 0.2
    o.ApplyStrokeMode = Enum.ApplyStrokeMode.Border
    o.Parent = parent
    return o
end

-- 投影 (纯 Frame 描边实现, 无贴图依赖, 不会出现透明底图残留)
local function shadow(parent, trans, circular)
    local BASE = trans or 0.72
    -- 用 3 层向外均匀扩散的描边模拟柔和投影
    for i = 1, 3 do
        local o = Instance.new("Frame")
        o.Name = "__shadow" .. i
        o.AnchorPoint = Vector2.new(0.5, 0.5)
        -- 圆形: 同心扩散; 矩形: 轻微下移模拟光源
        o.Position = circular and UDim2.new(0.5, 0, 0.5, 0)
                     or UDim2.new(0.5, 0, 0.5, 1 + i)
        o.Size = UDim2.new(1, i * 4, 1, i * 4)
        o.BackgroundColor3 = Color3.fromRGB(0, 0, 0)
        o.BackgroundTransparency = math.min(0.95, BASE + (i - 1) * 0.08)
        o.BorderSizePixel = 0
        o.ZIndex = (parent.ZIndex or 1) - 1
        o.Parent = parent

        local c = Instance.new("UICorner")
        -- 圆形对象用全圆角, 矩形对象用固定圆角
        c.CornerRadius = circular and UDim.new(1, 0) or UDim.new(0, 14 + i * 3)
        c.Parent = o
    end
    return nil
end

-- 渐变
local function gradient(parent, c1, c2, rot)
    local o = Instance.new("UIGradient")
    o.Color = ColorSequence.new{
        ColorSequenceKeypoint.new(0, c1),
        ColorSequenceKeypoint.new(1, c2)
    }
    o.Rotation = rot or 45
    o.Parent = parent
    return o
end

-- 内边距
local function padding(parent, all)
    local o = Instance.new("UIPadding")
    o.PaddingTop = UDim.new(0, all)
    o.PaddingBottom = UDim.new(0, all)
    o.PaddingLeft = UDim.new(0, all)
    o.PaddingRight = UDim.new(0, all)
    o.Parent = parent
    return o
end

-- 补间动画
local TweenService = game:GetService("TweenService")
local function tween(obj, time, props, style)
    local info = TweenInfo.new(
        time or 0.18,
        style or Enum.EasingStyle.Quad,
        Enum.EasingDirection.Out
    )
    local t = TweenService:Create(obj, info, props)
    t:Play()
    return t
end

-- 交互反馈 (悬停/按下)
local function interactive(btn, normalColor, hoverColor, pressColor)
    btn.AutoButtonColor = false
    -- 记录基准尺寸, 避免反复按下/离开造成偏移累积
    local baseSize = btn.Size
    local pressed = false

    btn.MouseEnter:Connect(function()
        if not pressed then
            tween(btn, 0.14, { BackgroundColor3 = hoverColor })
        end
    end)
    btn.MouseLeave:Connect(function()
        if pressed then
            pressed = false
            tween(btn, 0.12, { Size = baseSize })
        end
        tween(btn, 0.14, { BackgroundColor3 = normalColor })
    end)
    btn.MouseButton1Down:Connect(function()
        pressed = true
        if pressColor then
            tween(btn, 0.06, { BackgroundColor3 = pressColor })
        end
        tween(btn, 0.08, {
            Size = UDim2.new(baseSize.X.Scale, baseSize.X.Offset,
                             baseSize.Y.Scale, baseSize.Y.Offset + 2)
        })
    end)
    btn.MouseButton1Up:Connect(function()
        pressed = false
        tween(btn, 0.12, { BackgroundColor3 = hoverColor })
        tween(btn, 0.12, { Size = baseSize })
    end)
end

    end)

        local bpf = tonumber(bpfInput.Text)

        saveSettings()

    end

            setToggle(not MinigunAuto)

            countText.Text = "发射：" .. tostring(shotCount) .. "  (" .. remain .. "s)"
            -- 每 20 秒自动归零, 不持久化
            if elapsed >= RESET_INTERVAL then
                lastReset = os.clock()
                shotCount = 0
                -- 归零时闪一下作为提示
                countText.TextColor3 = THEME.accent
                task.delay(0.4, function()
                    countText.TextColor3 = THEME.subtext
                end)
            end
            task.wait(0.25)
        end
    end)

    -- ===== 核心射击 =====

    local RunService = game:GetService("RunService")

    local lastShot = 0

    RunService.Heartbeat:Connect(function()

        if not runLoop then return end

        if MinigunAuto and shootMode == "frame" then

            local intervalSec

            intervalSec = shootInterval / 1000

            if intervalSec <= 0 then

                for _ = 1, bulletsPerFrame do

                    pcall(function()

                        FirearmHandler = getsenv(game:GetService("Players").LocalPlayer.Character.FirearmHandler)

                        debug.setupvalue(FirearmHandler.BulletFire, 1, true)

                        FirearmHandler.PlayAnimation("PresentReady", "PresentFire")

                        FirearmHandler.BulletFire()

                    end)

                end

                shotCount = shotCount + bulletsPerFrame

                lastShot = tick()

            else

                local now = tick()

                if now - lastShot >= intervalSec then

                    for _ = 1, bulletsPerFrame do

                        pcall(function()

                            FirearmHandler = getsenv(game:GetService("Players").LocalPlayer.Character.FirearmHandler)

                            debug.setupvalue(FirearmHandler.BulletFire, 1, true)

                            FirearmHandler.PlayAnimation("PresentReady", "PresentFire")

                            FirearmHandler.BulletFire()

                        end)

                    end

                    shotCount = shotCount + bulletsPerFrame

                    lastShot = now

                end

            end

        end

    end)

    task.spawn(function()

        while runLoop do

            if MinigunAuto and shootMode == "wait" then

                local intervalSec

                intervalSec = shootInterval / 1000

                if intervalSec <= 0 then wait() else wait(intervalSec) end

                for _ = 1, bulletsPerFrame do

                    pcall(function()

                        FirearmHandler = getsenv(game:GetService("Players").LocalPlayer.Character.FirearmHandler)

                        debug.setupvalue(FirearmHandler.BulletFire, 1, true)

                        FirearmHandler.PlayAnimation("PresentReady", "PresentFire")

                        FirearmHandler.BulletFire()

                    end)

                end

                shotCount = shotCount + bulletsPerFrame

            else

                wait()

            end

        end

    end)

end

startScript()
