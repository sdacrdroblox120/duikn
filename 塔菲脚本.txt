-- ===== 密钥验证 =====

local function verifyKey()

    if keyInput.Text == SECRET_KEY then

        keyGui:Destroy()

        startScript()

    else

        keyInput.Text = ""

        keyInput.PlaceholderText = "密钥错误，请重试"

        keyInput.BackgroundColor3 = Color3.fromRGB(80, 30, 30)

        task.wait(1)

        keyInput.BackgroundColor3 = Color3.fromRGB(40, 40, 50)

    end

end

keyButton.MouseButton1Click:Connect(verifyKey)

keyInput.FocusLost:Connect(function(enterPressed)

    if enterPressed then verifyKey() end

end)

-- ===== 主脚本 =====

    local function saveSettings()

        local data = {

            shootInterval = shootInterval,

            mode = mode,

            bulletsPerFrame = bulletsPerFrame,

            shootMode = shootMode

        }

        pcall(function()

            writefile("minigun_settings.json", game:GetService("HttpService"):JSONEncode(data))

        end)

    end

    local function loadSettings()

        pcall(function()

            if readfile and isfile and isfile("minigun_settings.json") then

                local data = game:GetService("HttpService"):JSONDecode(readfile("minigun_settings.json"))

                if data.shootInterval then shootInterval = data.shootInterval end

                if data.mode then mode = data.mode end

                if data.bulletsPerFrame then bulletsPerFrame = data.bulletsPerFrame end

                if data.shootMode then shootMode = data.shootMode end

            end

        end)

    end

    -- ===== 悬浮球 =====

    local floatingBall = Instance.new("TextButton")

    floatingBall.Size = UDim2.new(0, 50, 0, 50)

    floatingBall.Position = UDim2.new(0.02, 0, 0.6, 0)

    floatingBall.BackgroundColor3 = Color3.fromRGB(30, 30, 45)

    floatingBall.Text = "⚙"

    floatingBall.TextColor3 = Color3.fromRGB(100, 180, 255)

    floatingBall.TextSize = 24

    floatingBall.Font = Enum.Font.GothamBold

    floatingBall.AutoButtonColor = false

    floatingBall.Visible = false

    floatingBall.Parent = screenGui

    local ballCorner = Instance.new("UICorner")

    ballCorner.CornerRadius = UDim.new(1, 0)

    ballCorner.Parent = floatingBall

    local ballStroke = Instance.new("UIStroke")

    ballStroke.Color = Color3.fromRGB(100, 150, 255)

    ballStroke.Thickness = 2

    ballStroke.Transparency = 0.3

    ballStroke.Parent = floatingBall

    local ballGradient = Instance.new("UIGradient")

    ballGradient.Color = ColorSequence.new{

        ColorSequenceKeypoint.new(0, Color3.fromRGB(50, 50, 80)),

        ColorSequenceKeypoint.new(1, Color3.fromRGB(20, 20, 35))

    }

    ballGradient.Rotation = 45

    ballGradient.Parent = floatingBall

    local ballDragging = false

    local ballDragStart = nil

    local ballStartPos = nil

    floatingBall.InputBegan:Connect(function(input)

        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then

            ballDragging = true

            ballDragStart = input.Position

            ballStartPos = floatingBall.Position

            input.Changed:Connect(function()

                if input.UserInputState == Enum.UserInputState.End then

                    ballDragging = false

                end

            end)

        end

    end)

    floatingBall.InputChanged:Connect(function(input)

        if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then

            if ballDragging then

                local delta = input.Position - ballDragStart

                floatingBall.Position = UDim2.new(

                    ballStartPos.X.Scale, ballStartPos.X.Offset + delta.X,

                    ballStartPos.Y.Scale, ballStartPos.Y.Offset + delta.Y

                )

            end

        end

    end)

    local function addHoverEffect(btn, normalColor, hoverColor)

        btn.MouseEnter:Connect(function()

            btn.BackgroundColor3 = hoverColor

        end)

        btn.MouseLeave:Connect(function()

            if btn == toggleButton then

                if MinigunAuto then

                    btn.BackgroundColor3 = Color3.fromRGB(50, 180, 80)

                else

                    btn.BackgroundColor3 = Color3.fromRGB(80, 80, 80)

                end

            else

                btn.BackgroundColor3 = normalColor

            end

        end)

    end

    addHoverEffect(toggleButton, Color3.fromRGB(80, 80, 80), Color3.fromRGB(110, 110, 110))

    addHoverEffect(closeButton, Color3.fromRGB(180, 50, 50), Color3.fromRGB(220, 70, 70))

    addHoverEffect(minButton, Color3.fromRGB(60, 60, 60), Color3.fromRGB(90, 90, 90))

    addHoverEffect(modeButton, Color3.fromRGB(50, 180, 80), Color3.fromRGB(70, 200, 100))

    addHoverEffect(shootModeButton, Color3.fromRGB(50, 120, 180), Color3.fromRGB(70, 140, 200))

    local minScale = 0.5

    local maxScale = 2.0

    local scaleStep = 0.1

    local function updateScale()

        uiScale.Scale = math.clamp(uiScale.Scale, minScale, maxScale)

    end

    zoomInButton.MouseButton1Click:Connect(function()

        uiScale.Scale = uiScale.Scale + scaleStep

        updateScale()

    end)

    zoomOutButton.MouseButton1Click:Connect(function()

        uiScale.Scale = uiScale.Scale - scaleStep

        updateScale()

    end)

    local function updateModeButton()

        if mode == 1 then

            modeButton.Text = "毫秒"

            intervalLabel.Text = "射速(毫秒):"

            modeButton.BackgroundColor3 = Color3.fromRGB(50, 180, 80)

        else

            modeButton.Text = "微秒"

            intervalLabel.Text = "射速(微秒):"

            modeButton.BackgroundColor3 = Color3.fromRGB(50, 120, 180)

        end

    end

    modeButton.MouseButton1Click:Connect(function()

        if mode == 1 then mode = 2 else mode = 1 end

        updateModeButton()

        saveSettings()

    end)

    local function updateShootModeButton()

        if shootMode == "frame" then

            shootModeButton.Text = "帧模式"

            shootModeButton.BackgroundColor3 = Color3.fromRGB(50, 120, 180)

        else

            shootModeButton.Text = "wait模式"

            shootModeButton.BackgroundColor3 = Color3.fromRGB(180, 120, 50)

        end

    end

    shootModeButton.MouseButton1Click:Connect(function()

        if shootMode == "frame" then shootMode = "wait" else shootMode = "frame" end

        updateShootModeButton()

        saveSettings()

    end)

    local function applyInputs()

        local num = tonumber(intervalInput.Text)

        if num and num >= 0 then shootInterval = num end

        local bpf = tonumber(bpfInput.Text)

        if bpf and bpf >= 0 then bulletsPerFrame = bpf end

        saveSettings()

    end

    intervalInput.FocusLost:Connect(function(ep) if ep then applyInputs() end end)

    bpfInput.FocusLost:Connect(function(ep) if ep then applyInputs() end end)

    toggleButton.MouseButton1Click:Connect(function()

        MinigunAuto = not MinigunAuto

        if MinigunAuto then

            toggleButton.Text = "自动射击：开"

            toggleButton.BackgroundColor3 = Color3.fromRGB(50, 180, 80)

            statusDot.BackgroundColor3 = Color3.fromRGB(50, 220, 80)

        else

            toggleButton.Text = "自动射击：关"

            toggleButton.BackgroundColor3 = Color3.fromRGB(80, 80, 80)

            statusDot.BackgroundColor3 = Color3.fromRGB(200, 50, 50)

        end

    end)

    closeButton.MouseButton1Click:Connect(function()

        MinigunAuto = false

        runLoop = false

        saveSettings()

        screenGui:Destroy()

    end)

    minButton.MouseButton1Click:Connect(function()

        mainFrame.Visible = false

        floatingBall.Visible = true

    end)

    floatingBall.MouseButton1Click:Connect(function()

        if not ballDragging then

            mainFrame.Visible = true

            floatingBall.Visible = false

        end

    end)

    if MinigunAuto then

        toggleButton.Text = "自动射击：开"

        toggleButton.BackgroundColor3 = Color3.fromRGB(50, 180, 80)

        statusDot.BackgroundColor3 = Color3.fromRGB(50, 220, 80)

    else

        toggleButton.Text = "自动射击：关"

        toggleButton.BackgroundColor3 = Color3.fromRGB(80, 80, 80)

        statusDot.BackgroundColor3 = Color3.fromRGB(200, 50, 50)

    end

    updateModeButton()

    updateShootModeButton()

    intervalInput.Text = tostring(shootInterval)

    bpfInput.Text = tostring(bulletsPerFrame)

    -- ===== 核心射击 =====

    local RunService = game:GetService("RunService")

    local lastShot = 0

    RunService.Heartbeat:Connect(function()

        if not runLoop then return end

        if MinigunAuto and shootMode == "frame" then

            local intervalSec

            if mode == 1 then intervalSec = shootInterval / 1000

            else intervalSec = shootInterval / 1000000 end

            if intervalSec <= 0 then

                for _ = 1, bulletsPerFrame do

                    pcall(function()

                        FirearmHandler = getsenv(game:GetService("Players").LocalPlayer.Character.FirearmHandler)

                        debug.setupvalue(FirearmHandler.BulletFire, 1, true)

                        FirearmHandler.PlayAnimation("PresentReady", "PresentFire")

                        FirearmHandler.BulletFire()

                    end)

                end

                lastShot = tick()

            else

                local now = tick()

                if now - lastShot >= intervalSec then

                    for _ = 1, bulletsPerFrame do

                        pcall(function()

                            FirearmHandler = getsenv(game:GetService("Players").LocalPlayer.Character.FirearmHandler)

                            debug.setupvalue(FirearmHandler.BulletFire, 1, true)

                            FirearmHandler.PlayAnimation("PresentReady", "PresentFire")

                            FirearmHandler.BulletFire()

                        end)

                    end

                    lastShot = now

                end

            end

        end

    end)

    task.spawn(function()

        while runLoop do

            if MinigunAuto and shootMode == "wait" then

                local intervalSec

                if mode == 1 then intervalSec = shootInterval / 1000

                else intervalSec = shootInterval / 1000000 end

                if intervalSec <= 0 then wait() else wait(intervalSec) end

                for _ = 1, bulletsPerFrame do

                    pcall(function()

                        FirearmHandler = getsenv(game:GetService("Players").LocalPlayer.Character.FirearmHandler)

                        debug.setupvalue(FirearmHandler.BulletFire, 1, true)

                        FirearmHandler.PlayAnimation("PresentReady", "PresentFire")

                        FirearmHandler.BulletFire()

                    end)

                end

            else

                wait()

            end

        end

    end)

end
